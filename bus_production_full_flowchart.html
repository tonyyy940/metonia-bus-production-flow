        A3["0.3 Generate Bill of Materials from approved drawing"]

        B9["1.9 Issue to production via Material Issue Note"]
        DOC_PO[/"Purchase Order + GRN"/]
        DOC_SNCR[/"Supplier Non-Conformance Report"/]
    end

    B1 --> B2 --> B3 --> B4 --> DOC_PO --> B5 --> B6 --> DB
    DB -- No --> B7 --> DOC_SNCR

    DB -- Yes --> B8 --> B9 --> C1

    %% ================= PHASE 2 =================
    subgraph PH2["PHASE 2 — Chassis Prep Bay"]
        direction TB

        subgraph P21["2.1 Chassis-Cab Reception and Intake Inspection"]
            C1["2.1.1 Receive complete chassis-cab unit"]
            C2["2.1.2 Verify chassis/VIN and engine number vs documents"]
            C3["2.1.3 Intake inspection: odometer, condition, fluids, tyres, battery"]
            C4["2.1.4 Photograph baseline condition, all sides and interior"]
            C5["2.1.5 Register chassis, assign job number, allocate bay"]
            QC1{{"QC HOLD: Chassis Intake Acceptance"}}
        end

        subgraph P22["2.2 Pre-Removal Preparation and Isolation"]
            C6["2.2.1 Position on level ground/hoist, park brake, chock wheels"]
            C7["2.2.2 Isolate power: battery negative then positive, lockout/tagout"]
            C8["2.2.3 Depressurise/drain air reservoirs and cab heater coolant circuit"]
            C9["2.2.4 Tag and photograph all harnesses, hoses, cables at cab interface"]
        end

        DCAB{{"Construction type?"}}

        subgraph P23["2.3 Cabin Disconnection"]
            C10["2.3.a Disconnect cab harness at main multi-pin connectors"]
            C11["2.3.b Disconnect accelerator, clutch, gear shift, brake, park brake linkages"]
            C12["2.3.c Disconnect steering column coupling from steering box"]
            C13["2.3.d Recover refrigerant, disconnect heater hoses and A/C lines"]
            C14["2.3.e Disconnect exhaust from cab-mounted brackets"]
            C15["2.3.f Disconnect washer lines, horn, cab-mounted lighting"]
            C16["2.3.g Remove cab-to-chassis mounting fasteners, record torques"]
            C17["2.3.h Release cab tilt mechanism and hinge pins - COE chassis"]
            QC2{{"QC HOLD: Pre-Lift Disconnection Check by 2nd technician"}}
        end

        subgraph P24["2.4 Cabin Lifting and Removal"]
            C18["2.4.1 Attach rated sling/chain and spreader bar at lift points"]
            C19["2.4.2 Position overhead crane, gantry or rated forklift attachment"]
            C20["2.4.3 Slow controlled vertical lift with spotter watching for snags"]
            C21["2.4.4 Lift clear and transfer cab to storage cradle/stand"]
            C22["2.4.5 Tag stored cab with chassis/job number, route per BOM"]
            QC3{{"SAFETY HOLD: Lift Sign-Off - rigging, load rating, clearance"}}
        end

        subgraph P25["2.5 Bare-Chassis Post-Removal Inspection and Hand-Off"]
            C23["2.5.1 Inspect frame rails: damage, corrosion, straightness"]
            C24["2.5.2 Verify wheelbase, frame width, front/rear overhang vs drawing"]
            C25["2.5.3 Cap and blank all open fuel, brake/air and electrical lines"]
            C26["2.5.4 Re-route/extend chassis wiring to new driver compartment location"]
            C27["2.5.5 Fit protective covers over engine bay, radiator, mechanicals"]
            C28["2.5.6 Degrease and clean chassis top flange for sub-frame mounting"]
            C29["2.5.7 Update tracker: Chassis-Cab to Bare Chassis"]
            QC4{{"QC HOLD: Bare-Chassis Acceptance - gate to Phase 3"}}
        end
    end

    end
